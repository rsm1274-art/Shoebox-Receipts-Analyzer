Why the original needed an API—and how we replace it locally:

The cloud showcase version sends receipt images to Google's servers for multimodal JSON extraction. To run 100% offline on consumer hardware (Mac M-Series, NVIDIA GPU, or modern CPU), you can either use **this web app directly** (which runs Tesseract WASM + Local Rules or connects to local Ollama), or run the standalone Python script below.

1. Install Local Ollama Vision Model (Terminal)

# Install Ollama from https://ollama.com, then pull a local vision model:
ollama pull llama3.2-vision
# Allow browser/local connections and start server:
OLLAMA_ORIGINS=* ollama serve

2. Standalone Python Script (`local_shoebox_analyzer.py`)

import base64
import json
import requests
import streamlit as st

OLLAMA_URL = "http://localhost:11434/api/generate"
MODEL_NAME = "llama3.2-vision"

SYSTEM_PROMPT = """
Extract receipt data into strict JSON with this schema:
{
  "merchant": "string",
  "date": "YYYY-MM-DD",
  "category": "Meals & Entertainment | Office Supplies | Travel & Fuel | Groceries | Utilities",
  "items": [{"name": "string", "qty": 1, "price": 0.00, "category": "string"}],
  "subtotal": 0.00,
  "tax": 0.00,
  "tip": 0.00,
  "total": 0.00
}
Return ONLY valid JSON.
"""

def analyze_receipt_locally(image_bytes):
    b64_img = base64.b64encode(image_bytes).decode("utf-8")
    payload = {
        "model": MODEL_NAME,
        "prompt": SYSTEM_PROMPT,
        "images": [b64_img],
        "format": "json",
        "stream": False
    }
    res = requests.post(OLLAMA_URL, json=payload, timeout=120)
    data = json.loads(res.json()["response"])
    return audit_receipt_anomalies(data)

def audit_receipt_anomalies(receipt):
    anomalies = []
    items_sum = round(sum(i.get("price", 0) * i.get("qty", 1) for i in receipt.get("items", [])), 2)
    subtotal = receipt.get("subtotal", items_sum)
    expected_total = round(subtotal + receipt.get("tax", 0) + receipt.get("tip", 0), 2)

    if abs(items_sum - subtotal) > 0.05:
        anomalies.append(f"Line items sum (${items_sum}) != Subtotal (${subtotal})")
    if abs(expected_total - receipt.get("total", 0)) > 0.05:
        anomalies.append(f"Math discrepancy: Expected ${expected_total}, charged ${receipt.get('total')}")
    if subtotal > 0 and (receipt.get("tip", 0) / subtotal) > 0.22:
        anomalies.append(f"Excessive tip: {round((receipt['tip']/subtotal)*100, 1)}% exceeds 22% policy")

    receipt["anomalies"] = anomalies
    return receipt

st.title("Shoebox Receipts Analyzer (100% Local Ollama)")
uploaded = st.file_uploader("Upload Receipt Image", type=["png", "jpg", "jpeg"])
if uploaded and st.button("Analyze Locally"):
    with st.spinner("Running local vision model..."):
        result = analyze_receipt_locally(uploaded.read())
        st.json(result)
